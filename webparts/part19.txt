function addPeriod(dateStr,frequency,n){
  const [y,m,d]=dateStr.split('-').map(Number);
  if(frequency==='weekly'){
    const dt=new Date(y,m-1,d); dt.setDate(dt.getDate()+7*n);
    return `${dt.getFullYear()}-${String(dt.getMonth()+1).padStart(2,'0')}-${String(dt.getDate()).padStart(2,'0')}`;
  }
  if(frequency==='yearly'){
    const yy=y+n, mm=m, last=new Date(yy,mm,0).getDate(), dd=Math.min(d,last);
    return `${yy}-${String(mm).padStart(2,'0')}-${String(dd).padStart(2,'0')}`;
  }
  const total=(m-1)+n, yy=y+Math.floor(total/12), mm=(total%12)+1, last=new Date(yy,mm,0).getDate(), dd=Math.min(d,last);
  return `${yy}-${String(mm).padStart(2,'0')}-${String(dd).padStart(2,'0')}`;
}
function fileToData(file){
  return new Promise((resolve,reject)=>{
    if(!file) return resolve(null);
    if(file.size>1800000) return reject(new Error('O anexo deve ter no máximo 1,8 MB nesta versão local.'));
    const r=new FileReader(); r.onload=()=>resolve({name:file.name,data:r.result}); r.onerror=reject; r.readAsDataURL(file);
  });
}
