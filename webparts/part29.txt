function saveNewReceipt(){
  const clientId=Number(val('rec_new_client'));
  const competence=val('rec_new_comp');
  const issueDate=val('rec_new_date');
  const value=parseMoneyBR(val('rec_new_value'));

  const c=clients.find(x=>Number(x.id)===clientId);

  if(!c)return alert('Selecione o cliente.');
  if(!competence)return alert('Informe a competência.');
  if(!issueDate)return alert('Informe a data do recibo.');
  if(!(value>0))return alert('Informe um valor válido.');

  const year=String(competence).slice(0,4);

  const row=normalizeReceipt({
    id:Date.now(),
    sourceBoletoId:'',
    clientId:c.id,
    client:c.name,
    competence,
    issueDate,
    value,
    description:'Honorários administrativos',
    receiptNumber:nextReceiptNumber(year),
    status:'pendente'
  });

  receipts.push(row);

  saveData('receipts',receipts);

  closeModal();
  renderAll();

  alert(
    `Recibo criado ✅\n\n`+
    `Nº: ${row.receiptNumber}\n`+
    `Cliente: ${row.client}\n`+
    `Valor: ${money(row.value)}`
  );
}

function saveReceiptEdit(id){
  const x=receipts.find(r=>r.id===id);if(!x)return;
  const number=val('rec_number').trim(),comp=val('rec_comp'),date=val('rec_date'),value=parseMoneyBR(val('rec_value'));
  if(!number)return alert('Informe o número do recibo.');
  if(!comp)return alert('Informe a competência.');
  if(!date)return alert('Informe a data do recibo.');
  if(!(value>0))return alert('Informe um valor válido.');
  if(receipts.some(r=>r.id!==id&&String(r.receiptNumber).toUpperCase()===number.toUpperCase()))return alert('Já existe um recibo com esse número.');
  x.receiptNumber=number;x.competence=comp;x.issueDate=date;x.value=value;x.status='pendente';
  saveData('receipts',receipts);closeModal();renderAll();
}
function deleteReceipt(id){
  const x=receipts.find(r=>r.id===id);if(!x)return;
  if(!confirm(`Excluir o recibo ${x.receiptNumber||''} de ${x.client}?\n\nIsso não exclui o boleto de origem.`))return;
  receipts=receipts.filter(r=>r.id!==id);saveData('receipts',receipts);renderAll();
}
