fse'):'nfse'
  });

  if(!obj.name)return alert('Informe o nome.');

  if(type==='client'){
    clients=id?clients.map(x=>x.id===id?obj:x):[...clients,obj];
    saveData('clients',clients);
  }else{
    suppliers=id?suppliers.map(x=>x.id===id?obj:x):[...suppliers,obj];
    saveData('suppliers',suppliers);
  }

  closeModal();
  renderAll();
}

function nfseMonthLabel(comp){
  const m=String(comp||'').match(/^(\d{4})-(\d{2})$/);
  return m?`${m[2]}/${m[1]}`:(comp||'-');
}
function nfseLastDay(comp){
  const m=String(comp||'').match(/^(\d{4})-(\d{2})$/);
  if(!m)return today();
  const d=new Date(Number(m[1]),Number(m[2]),0);
  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;
}
function nfseSetting(comp){return nfseSettings[comp]||{};}
function nfseAliquota(comp,row){return Number(row?.aliquotaPct||nfseSetting(comp).aliquotaPct||0);}
function nfseServiceDate(comp,row){return row?.serviceDate||nfseSetting(comp).serviceDate||nfseLastDay(comp);}
function nfseClientIbge(c){
  if(c?.cityIbge&&/^\d{7}$/.test(String(c.cityIbge)))return String(c.cityIbge);
  if(looseName(c?.city)==='SAOJOSEDORIOPRETO')return '3549805';
  return '';
}
function nfseClientCheck(c){
  if(!c)return 'Cliente não encontrado no cadastro.';
  const d=String(c.doc||'').replace(/\D/g,'');
  const missing=[];
  if(![11,14].includes(d.length))missing.push('CNPJ/CPF');
  if(!c.name)missing.push('Razão Social');
  if(!c.street)missing.push('endereço');
  if(!c.number)missing.push('número');
  if(!c.district)missing.push('bairro');
  if(!c.zip||String(c.zip).replace(/\D/g,'').length!==8)missing.push('CEP');
  if(!c.city)missing.push('cidade');
  if(!c.state)missing.push('UF');
  if(!nfseClientIbge(c))missing.push('código IBGE da cidade');
  return missing.length?`Complete no cliente: ${missing.join(', ')}.`:'';
}
function nfseTomadorFromClient(c){
  const err=nfseClientCheck(c); if(err)throw new Error(err);
  return {
    documento:String(c.doc||'').replace(/\D/g,''),nome:c.name,
    endereco:c.street,numero:c.number,complemento:c.complement||'',bairro:c.district,
    codigoMunicipio:nfseClientIbge(c),cidade:c.city,uf:c.state,cep:String(c.zip||'').replace(/\D/g,''),email:c.email||''
  };
}
function nfsePayload(row){
  const c=clients.find(x=>x.id===row.clientId)||findClientByLooseName(row.client);
  const aliquota=nfseAliquota(row.competence,row);
  if(!row.competence)throw new Error('Defina a competência.');
  if(!aliquota)throw new Error(`Configure a alíquota da competência ${nfseMonthLabel(row.competence)}.`);
  return {
    numeroRps:String(row.rpsNumber),
    numeroLote:Date.now(),
    competencia:nfseServiceDate(row.competence,row),
    dataEmissao:row.issueDate||today(),
    valor:Number(row.value||0),
    aliquotaPct:aliquota,
    discriminacao:row.description||'Administração de Condomínios.',
    tomador:nfseTomadorFromClient(c)
  };
}
function nfseCanEmit(row){
  if(row.status==='emitida_nfse')return 'Já emitida.';
  if(row.status==='cancelada_nfse')return 'NFS-e cancelada.';
  if(!row.competence)return 'Sem competência.';
  if(!nfseAliquota(row.competence,row))return 'Alíquota do mês não configurada.';
  const c=clients.find(x=>x.id===row.clientId)||findClientByLooseName(row.client);
  return nfseClientCheck(c);
}
function nextRpsNumber(){
  const lastSaved = Number(
    localStorage.getItem('brcondos_giss_last_rps') || 0
  );

  const used = nfse
    .map(x => Number(x.rpsNumber))
    .filter(n =>
      Number.isFinite(n) &&
      n > 0 &&
      n < 1000000
    );

  const maior = Math.max(
    lastSaved,
    ...(used.length ? used : [0])
  );

  return String(maior + 1);
}
