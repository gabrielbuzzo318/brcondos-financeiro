function reportMatchesSearch(values,q){
  if(!q) return true;
  const hay=normalizeSearchText(values.filter(Boolean).join(' '));
  const query=normalizeSearchText(q);
  return hay.includes(query) || valueSearchVariants(query).some(v=>hay.includes(normalizeSearchText(v)));
}
function reportDateOk(date,from,to){
  if(!date) return !from && !to;
  return (!from || date>=from) && (!to || date<=to);
}
