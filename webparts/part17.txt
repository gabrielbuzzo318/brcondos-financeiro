(function mergeImportedPayables(){
  // Remove apenas os dois registros demonstrativos das versões iniciais.
  payables=payables.filter(p=>!(
    (p.id===1 && p.supplier==='Alpha Serviços') ||
    (p.id===2 && p.supplier==='Tech Predial')
  ));

  const existingKeys=new Set(payables.map(p=>p.sourceKey).filter(Boolean));
  let nextId=2000000;

  BRCONDOS_PAYABLE_IMPORT.forEach(item=>{
    if(!existingKeys.has(item.sourceKey)){
      payables.push(normalizePayable({...item,id:nextId++}));
      existingKeys.add(item.sourceKey);
    }
  });

  saveData('payables',payables);
})();
