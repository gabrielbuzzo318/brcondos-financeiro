kpi-value">${success}</div></div>
      <div class="card" style="border-top:3px solid var(--danger)"><div class="kpi-label">FALHAS</div><div class="kpi-value">${failures.length}</div></div>
      <div class="card accent-slate"><div class="kpi-label">IGNORADOS</div><div class="kpi-value">${skipped.length}</div></div>
    </div>
    <div class="notice"><b>PDFs não foram baixados.</b> Você continua podendo abrir o PDF oficial individualmente quando quiser.</div>
    ${failHtml}
    ${skipHtml}
    <div style="display:flex;justify-content:flex-end;margin-top:18px"><button class="btn primary" onclick="closeModal()">Fechar</button></div>
  `);
}

async function consultSicrediBoleto(id){
  const b=boletos.find(x=>x.id===id);
  if(!b?.sicrediNossoNumero)return alert('Nosso Número não disponível.');
  try{
    const response=await fetch(`/api/boletos/${encodeURIComponent(b.sicrediNossoNumero)}`);
    const data=await response.json();
    if(!response.ok) throw new Error(data.error||'Falha na consulta.');
    b.sicrediResponse=data;
    saveData('boletos',boletos);
    alert(`Consulta Sicredi concluída.\n\nNosso Número: ${b.sicrediNossoNumero}\nStatus retornado: ${data.situacao || data.status || data.situacaoBoleto || 'consulte o JSON salvo no registro'}`);
  }catch(e){
    alert(`Erro na consulta Sicredi:\n${e.message}`);
  }
}

function payBoleto(id){
  const b=boletos.find(x=>x.id===id);
  if(!b)return;
  openModal('Baixa do boleto',`
    <div class="modal-grid">
      ${field('Cliente',`<input value="${esc(b.client)}" disabled>`)}
      ${field('Boleto',`<input value="${esc(b.docNumber)}" disabled>`)}
      ${field('Valor recebido',`<input value="${money(b.value)}" disabled>`)}
      ${field('Data da baixa / recebimento',`<input id="boleto_receipt_date" type="date" value="${b.receiptDate||today()}">`)}
    </div>
    <div class="notice" style="margin-top:14px">A data escolhida será a data da entrada no Fluxo de Caixa.</div>
    <div style="display:flex;justify-content:flex-end;gap:8px">
      <button class="btn" onclick="closeModal()">Cancelar</button>
      <button class="btn primary" onclick="confirmBoletoReceived(${id})">Confirmar recebimento</button>
    </div>`);
}
function confirmBoletoReceived(id){
  const d=val('boleto_receipt_date');
  if(!d)return alert('Informe a data da baixa.');
  boletos=boletos.map(x=>x.id===id?{...x,status:'recebido',receiptDate:d}:x);
  saveData('boletos',boletos);
  syncBoletoToFlow(id);
  closeModal();
  renderAll();
}
function syncBoletoToFlow(id){
  const b=boletos.find(x=>x.id===id);
  if(!b || b.flowId) return;
  const flowId=Date.now()+13;
  transactions.push({
    id:flowId,
    date:b.receiptDate||today(),
    type:'entrada',
    description:`Boleto recebido - ${b.description}`,
    category:'Recebimento de boletos',
    party:b.client||'',
    value:Number(b.value),
    status:'pago'
  });
  boletos=boletos.map(x=>x.id===id?{...x,flowId}:x);
  saveData('transactions',transactions);
  saveData('boletos',boletos);
}
