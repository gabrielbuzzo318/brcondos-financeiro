function payableEffectiveStatus(p){
  if(p.status==='pago' || p.status==='agendado') return p.status;
  if(p.due && p.due < today()) return 'vencido';
  return 'aberto';
}
function recurrenceBadge(p){
  if(!p.recurring) return '<span class="badge gray">Única</span>';
  const map={monthly:'Mensal',weekly:'Semanal',yearly:'Anual'};
  return `<span class="badge blue">${map[p.frequency]||'Recorrente'}</span>`;
}
