function accountOptions(type,current=''){
  const list=chartAccounts
    .filter(x=>!type || x.type===type)
    .sort((a,b)=>(a.code||'').localeCompare(b.code||''));
  let html='<option value="">Selecione</option>';
  html+=list.map(x=>`<option value="${esc(x.name)}" ${current===x.name?'selected':''}>${esc(x.code)} - ${esc(x.name)}</option>`).join('');
  return html;
}

function quickCreateAccount(type,targetSelectId){
  const name=prompt('Nome da nova conta:');
  if(!name)return;
  const group=prompt('Grupo (opcional):')||'';
  const code=prompt('Código (opcional, ex.: 2.11):')||'';
  const exists=chartAccounts.find(x=>x.name.trim().toLowerCase()===name.trim().toLowerCase());
  if(exists){
    alert('Essa conta já existe.');
    const sel=document.getElementById(targetSelectId);
    if(sel)sel.value=exists.name;
    return;
  }
  const obj={id:Date.now(),code,name:name.trim(),type,group:group.trim()};
  chartAccounts.push(obj);
  saveData('chartAccounts',chartAccounts);
  const sel=document.getElementById(targetSelectId);
  if(sel){
    sel.innerHTML=accountOptions(type,obj.name);
    sel.value=obj.name;
  }
}
function refreshTransactionAccountOptions(){
  const sel=document.getElementById('m_cat');
  const typeSel=document.getElementById('m_type');
  if(!sel||!typeSel)return;
  const current=sel.value;
  sel.innerHTML=accountOptions(typeSel.value,current);
}

function openChartAccount(id=null){
  const x=id?chartAccounts.find(a=>a.id===id):{code:'',name:'',type:'saida',group:''};
  openModal(id?'Editar conta':'Nova conta do plano',`
    <div class="modal-grid">
      ${field('Código',`<input id="pc_code" value="${esc(x.code||'')}" placeholder="Ex.: 2.11">`)}
      ${field('Tipo',`<select id="pc_type"><option value="entrada" ${x.type==='entrada'?'selected':''}>Entrada / Receita</option><option value="saida" ${x.type==='saida'?'selected':''}>Saída / Despesa</option></select>`)}
      ${field('Nome da conta',`<input id="pc_name" value="${esc(x.name||'')}" placeholder="Ex.: Honorários jurídicos">`)}
      ${field('Grupo',`<input id="pc_group" value="${esc(x.group||'')}" placeholder="Ex.: Despesas Administrativas">`)}
    </div>
    <div style="display:flex;justify-content:flex-end;gap:8px;margin-top:18px">
      <button class="btn" onclick="closeModal()">Cancelar</button>
      <button class="btn primary" onclick="saveChartAccount(${id||'null'})">Salvar</button>
    </div>`);
}
function saveChartAccount(id){
  const obj={
    id:id||Date.now(),
    code:val('pc_code'),
    name:val('pc_name'),
    type:val('pc_type'),
    group:val('pc_group')
  };
  if(!obj.name)return alert('Informe o nome da conta.');
  const duplicate=chartAccounts.some(x=>x.id!==id && x.name.trim().toLowerCase()===obj.name.trim().toLowerCase());
  if(duplicate)return alert('Já existe uma conta com esse nome.');
  if(id)chartAccounts=chartAccounts.map(x=>x.id===id?obj:x);else chartAccounts.push(obj);
  saveData('chartAccounts',chartAccounts);
  closeModal();renderAll();
}
function deleteChartAccount(id){
  const x=chartAccounts.find(a=>a.id===id);
  if(!x)return;
  if(!confirm(`Excluir a conta "${x.name}" do plano de contas? Os lançamentos existentes não serão apagados.`))return;
  chartAccounts=chartAccounts.filter(a=>a.id!==id);
  saveData('chartAccounts',chartAccounts);
  renderAll();
}
