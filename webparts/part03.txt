(function migrateBoletoDueMonthsOnce(){
  const migrationKey='brcondos_boleto_due_shift_20260825_v1';
  if(localStorage.getItem(migrationKey)==='done') return;

  boletos=boletos.map(b=>{
    if(!b.due) return b;
    const m=String(b.due).match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if(!m) return b;
    let year=Number(m[1]), month=Number(m[2]), day=m[3];

    if(month===7) month=8;
    else if(month===8) month=9;
    else return b;

    return {...b,due:`${year}-${String(month).padStart(2,'0')}-${day}`};
  });

  saveData('boletos',boletos);
  localStorage.setItem(migrationKey,'done');
})();

function looseName(v){
  return String(v||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toUpperCase().replace(/[^A-Z0-9]/g,'');
}

  if(!n) return null;

  if(exact) return exact;

    'ASSOCIACAOEPLENUM':'ASSOCIACAODOSMORADORESEPLENUMAMOEPLENUM',
