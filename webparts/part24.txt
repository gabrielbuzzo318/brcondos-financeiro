function filterSuppliersTable(){
  const q=normalizeSearchText(
    document.getElementById('supplier_search')?.value||''
  );

  const rows=[
    ...document.querySelectorAll('#view-fornecedores tbody tr[data-search]')
  ];

  let visible=0;

  rows.forEach(tr=>{
    const hay=normalizeSearchText(tr.dataset.search||'');
    const show=!q||hay.includes(q);

    tr.style.display=show?'':'none';
    if(show)visible++;
  });

  const result=document.getElementById('supplier_filter_result');
  if(result)result.textContent=`${visible} de ${rows.length} fornecedor(es)`;
}

function clearSuppliersFilter(){
  const input=document.getElementById('supplier_search');
  if(input)input.value='';
  filterSuppliersTable();
}

async function lookupPartyForm(){
  const doc=val('p_doc');
  const notice=document.getElementById('partyLookupNotice');
  if(cleanCnpj(doc).length!==14){notice.textContent='Digite um CNPJ válido com 14 números.';return}
  notice.textContent='Consultando CNPJ...';
  try{
    const data=await fetchCnpjPublic(doc);
    document.getElementById('p_name').value=data.razao_social||'';
    document.getElementById('p_trade').value=data.nome_fantasia||'';
    document.getElementById('p_status').value=data.descricao_situacao_cadastral||data.situacao_cadastral||'';
    document.getElementById('p_email').value=data.email||'';
    document.getElementById('p_phone').value=formatPhone(data.ddd_telefone_1||'');
    document.getElementById('p_zip').value=formatCep(data.cep||'');
    document.getElementById('p_street').value=[data.descricao_tipo_de_logradouro,data.logradouro].filter(Boolean).join(' ').replace(/\s+/g,' ').trim();
    document.getElementById('p_number').value=data.numero||'';
    document.getElementById('p_complement').value=data.complemento||'';
    document.getElementById('p_district').value=data.bairro||'';
    document.getElementById('p_city').value=data.municipio||'';
    document.getElementById('p_state').value=data.uf||'';
    if(looseName(data.municipio)==='SAOJOSEDORIOPRETO')document.getElementById('p_city_ibge').value='3549805';
    document.getElementById('p_cnae').value=data.cnae_fiscal?`${data.cnae_fiscal} - ${data.cnae_fiscal_descricao||''}`.trim():'';
    notice.textContent='Dados carregados. Confira e clique em Salvar.';
  }catch(e){
    notice.textContent=`Não foi possível consultar este CNPJ. ${e.message||''}`;
  }
}
