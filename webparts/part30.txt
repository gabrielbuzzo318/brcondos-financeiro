function normalizeSearchText(v){
  return String(v||'')
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'')
    .toLowerCase().trim();
}
function valueSearchVariants(v){
  const s=String(v||'').trim();
  return [s, s.replace(/\./g,'').replace(',','.'), s.replace(/[^\d,.-]/g,'')].filter(Boolean);
}

function filterBoletosTable(){
  const q=normalizeSearchText(document.getElementById('boleto_search')?.value||'');
  const from=document.getElementById('boleto_from')?.value||'';
  const to=document.getElementById('boleto_to')?.value||'';
  const status=document.getElementById('boleto_status')?.value||'';

  const rows=[...document.querySelectorAll('#view-boletos tbody tr[data-search]')];
  let visible=0;

  rows.forEach(tr=>{
    const hay=normalizeSearchText(tr.dataset.search||'');
    const date=tr.dataset.date||'';
    const st=tr.dataset.status||'';

    const matchesText=!q || hay.includes(q) || valueSearchVariants(q).some(v=>hay.includes(normalizeSearchText(v)));
    const matchesFrom=!from || (!!date && date>=from);
    const matchesTo=!to || (!!date && date<=to);
    const matchesStatus=!status || st===status;

    const show=matchesText && matchesFrom && matchesTo && matchesStatus;
    tr.style.display=show?'':'none';
    if(show) visible++;
  });

  const box=document.getElementById('boleto_filter_result');
  if(box) box.textContent=`${visible} de ${rows.length} boleto(s)`;
}

function clearBoletosFilters(){
  ['boleto_search','boleto_from','boleto_to','boleto_status'].forEach(id=>{
    const el=document.getElementById(id);
    if(el) el.value='';
  });
  filterBoletosTable();
}
