async function saveReimbursement(id){
  const old=id?reimbursements.find(x=>x.id===id):null;
  let attachmentName=old?.attachmentName||'', attachmentData=old?.attachmentData||'';

  try{
    const f=document.getElementById('rb_attachment').files[0];
    const payload=await fileToData(f);
    if(payload){attachmentName=payload.name;attachmentData=payload.data}
  }catch(e){
    return alert(e.message);
  }

  const obj={
    id:id||Date.now(),
    date:val('rb_date'),
    paidBy:val('rb_paidby'),
    reimbursedBy:val('rb_reimbursedby'),
    description:val('rb_desc'),
    category:val('rb_cat'),
    value:Number(val('rb_value')),
    requestDate:val('rb_request'),
    expectedDate:val('rb_expected'),
    receivedDate:old?.receivedDate||'',
    receiptMethod:val('rb_method'),
    status:val('rb_status'),
    notes:val('rb_notes'),
    attachmentName,attachmentData,
    flowId:old?.flowId||null
  };

  if(!obj.date||!obj.paidBy||!obj.reimbursedBy||!obj.description||!obj.value)
    return alert('Preencha data, quem pagou, quem deve reembolsar, descrição e valor.');

  if(id) reimbursements=reimbursements.map(x=>x.id===id?obj:x);
  else reimbursements.push(obj);

  saveData('reimbursements',reimbursements);

  if(obj.status==='recebido'){
    reimbursements=reimbursements.map(x=>x.id===obj.id?{...x,receivedDate:x.receivedDate||today()}:x);
    saveData('reimbursements',reimbursements);
    syncReimbursementToFlow(obj.id);
  }

  closeModal();
  renderAll();
}

function requestReimbursement(id){
  reimbursements=reimbursements.map(x=>x.id===id?{
    ...x,
    status:'solicitado',
    requestDate:x.requestDate||today()
  }:x);
  saveData('reimbursements',reimbursements);
  renderAll();
}

function receiveReimbursement(id){
  if(!confirm('Marcar este reembolso como recebido e lançar a entrada no Fluxo de Caixa?')) return;
  reimbursements=reimbursements.map(x=>x.id===id?{
    ...x,
    status:'recebido',
    receivedDate:today()
  }:x);
  saveData('reimbursements',reimbursements);
  syncReimbursementToFlow(id);
  renderAll();
}

function syncReimbursementToFlow(id){
  const r=reimbursements.find(x=>x.id===id);
  if(!r||r.flowId)return;

  const flowId=Date.now()+7;
  transactions.push({
    id:flowId,
    date:r.receivedDate||today(),
    type:'entrada',
    description:`Reembolso recebido - ${r.description}`,
    category:r.category||'Reembolsos',
    party:r.reimbursedBy||'',
    value:Number(r.value),
    status:'pago'
  });

  reimbursements=reimbursements.map(x=>x.id===id?{...x,flowId}:x);
  saveData('reimbursements',reimbursements);
  saveData('transactions',transactions);
}

function openReimbursementAttachment(id){
  const r=reimbursements.find(x=>x.id===id);
  if(!r?.attachmentData)return alert('Este reembolso não possui anexo.');
  const a=document.createElement('a');
  a.href=r.attachmentData;
  a.download=r.attachmentName||'reembolso';
  document.body.appendChild(a);
  a.click();
  a.remove();
}

function deleteReimbursement(id){
  if(!confirm('Excluir este reembolso?'))return;
  reimbursements=reimbursements.filter(x=>x.id!==id);
  saveData('reimbursements',reimbursements);
  renderAll();
}
