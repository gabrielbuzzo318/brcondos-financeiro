){
    return alert('Não há NFS-e aptas para emissão nos resultados atuais.');
  }

  const total=eligible.reduce((a,b)=>a+Number(b.value||0),0);

  if(!confirm(
    `EMISSÃO AUTOMÁTICA - GISS\n\n` +
    `NFS-e aptas: ${eligible.length}\n` +
    `Valor total: ${money(total)}\n` +
    `Ignoradas: ${skipped.length}\n\n` +
    `As notas serão emitidas UMA POR UMA automaticamente.\n\n` +
    `Confirmar?`
  )) return;

  openModal(
    'Emissão automática de NFS-e',
    `<div class="notice">
       <b>Não feche esta tela.</b>
       As NFS-e serão transmitidas individualmente para a Giss.
     </div>
     <div id="bulkProgressText">Preparando...</div>
     <div class="bulk-progress-track">
       <div id="bulkProgressBar" class="bulk-progress-bar"></div>
     </div>
     <div id="bulkProgressStats" class="subtle">
       Emitidas: 0 • Falhas: 0
     </div>
     <div id="bulkProgressLog" class="bulk-log">Iniciando...</div>`
  );

  let done=0;
  let success=0;
  let failed=0;

  for(const row of eligible){

    row.status='enviando_nfse';
    saveData('nfse',nfse);

    try{
      const payload=nfsePayload(row);

      const r=await fetch('/api/nfse/emitir',{
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body:JSON.stringify(payload)
      });

      const data=await r.json();

      applyGissResultToRows([row],data);

      if(row.status==='emitida_nfse'){
        success++;
      }else{
        failed++;
      }

      done++;

      updateBulkProgress(
        done,
        eligible.length,
        `${row.status==='emitida_nfse'?'✓':'✗'} RPS ${row.rpsNumber}${row.nfseNumber?' → NFS-e '+row.nfseNumber:''}`,
        success,
        failed
      );

    }catch(e){

      row.status='erro_nfse';
      row.lastError=e.message;
      saveData('nfse',nfse);

      failed++;
      done++;

      updateBulkProgress(
        done,
        eligible.length,
        `✗ RPS ${row.rpsNumber}: ${e.message}`,
        success,
        failed
      );
    }

    await sleepMs(700);
  }

  renderAll();

  openModal(
    'Emissão de NFS-e concluída',
    `<div class="cards grid" style="grid-template-columns:repeat(3,1fr)">
       <div class="card accent-green">
         <div class="kpi-label">EMITIDAS</div>
         <div class="kpi-value">${success}</div>
       </div>

       <div class="card" style="border-top:3px solid var(--danger)">
         <div class="kpi-label">FALHAS/PENDENTES</div>
         <div class="kpi-value">${failed}</div>
       </div>

       <div class="card accent-slate">
         <div class="kpi-label">IGNORADAS</div>
         <div class="kpi-value">${skipped.length}</div>
       </div>
     </div>

     <div style="display:flex;justify-content:flex-end;margin-top:18px">
       <button class="btn primary" onclick="closeModal()">Fechar</button>
     </div>`
  );
}

const receiptMonths=['JANEIRO','FEVEREIRO','MARÇO','ABRIL','MAIO','JUNHO','JULHO','AGOSTO','SETEMBRO','OUTUBRO','NOVEMBRO','DEZEMBRO'];
const receiptMonthsTitle=['Janeiro','Fevereiro','Março','Abril','Maio','Junho','Julho','Agosto','Setembro','Outubro','Novembro','Dezembro'];
function receiptCompetenceLabel(comp){
  const m=String(comp||'').match(/^([0-9]{4})-([0-9]{2})$/);
  if(!m)return String(comp||'-').toUpperCase();
  return `${receiptMonths[Number(m[2])-1]||m[2]}/${m[1]}`;
}
function receiptDateLong(date){
  const m=String(date||'').match(/^([0-9]{4})-([0-9]{2})-([0-9]{2})$/);
  if(!m)return formatDate(date);
  return `${Number(m[3])} de ${receiptMonthsTitle[Number(m[2])-1]||m[2]} de ${m[1]}`;
}
