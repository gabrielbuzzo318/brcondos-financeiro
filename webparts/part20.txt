    paymentMethod:val('ap_method'),

    notes:val('ap_notes'),

  if(id){
    payables=payables.map(x=>x.id===id?obj:x);

function confirmPayablePaid(id){
  const d=val('payable_payment_date');
  if(!d)return alert('Informe a data da baixa.');
  payables=payables.map(x=>x.id===id?{...x,status:'pago',paymentDate:d}:x);
  saveData('payables',payables);
  syncPayableToFlow(id);
  closeModal();
  renderAll();
}

function reversePayablePayment(id){
  const p=payables.find(x=>x.id===id);
  if(!p)return;

  if(!confirm('Estornar a baixa desta conta? Ela voltará para A pagar e a saída correspondente será removida do Fluxo de Caixa.')) return;

  if(p.flowId){
    transactions=transactions.filter(t=>t.id!==p.flowId);
    saveData('transactions',transactions);
  }

  payables=payables.map(x=>x.id===id?{
    ...x,
    status:'aberto',
    paymentDate:'',
    flowId:null
  }:x);

  saveData('payables',payables);
  renderAll();
}

function syncPayableToFlow(id){
  const p=payables.find(x=>x.id===id);
  if(!p || p.flowId) return;
  const flowId=Date.now()+1;
  transactions.push({
    id:flowId,date:p.paymentDate||today(),type:'saida',description:p.description,category:p.category||'Contas a pagar',
    party:p.supplier||'',value:Number(p.value),status:'pago'
  });
  payables=payables.map(x=>x.id===id?{...x,flowId}:x);
  saveData('transactions',transactions);saveData('payables',payables);
}
function delPayable(id){
  if(!confirm('Excluir esta conta a pagar?')) return;
  payables=payables.filter(x=>x.id!==id);
  saveData('payables',payables);renderAll();
}



function reimbursementStatusBadge(s){
  const map={
    a_solicitar:['A solicitar','gray'],
    solicitado:['Solicitado','blue'],
    vencido:['Vencido','red'],
    recebido:['Recebido','green'],
    cancelado:['Cancelado','red']
  };
  const x=map[s]||[s,'gray'];
  return `<span class="badge ${x[1]}">${x[0]}</span>`;
}
function reimbursementEffectiveStatus(r){
  if(r.status==='recebido'||r.status==='cancelado'||r.status==='a_solicitar') return r.status;
  if(r.expectedDate && r.expectedDate < today()) return 'vencido';
  return r.status||'solicitado';
}
