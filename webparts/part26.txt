function receiptDefaultIssueDate(competence,boletoDue=''){
  if(boletoDue)return boletoDue;
  const m=String(competence||'').match(/^(\d{4})-(\d{2})$/);
  if(!m)return today();
  let year=Number(m[1]), month=Number(m[2])+1;
  if(month===13){month=1;year++;}
  return `${year}-${String(month).padStart(2,'0')}-10`;
}
function receiptYear(row){
  return String(row?.competence||row?.issueDate||today()).slice(0,4) || String(new Date().getFullYear());
}
function nextReceiptNumber(year){
  year=String(year||new Date().getFullYear());
  const saved=Number(localStorage.getItem(`brcondos_receipt_last_${year}`)||0);
  const used=receipts.map(r=>{
    const m=String(r.receiptNumber||'').match(/^(\d+)\/(\d{4})$/);
    return m&&m[2]===year?Number(m[1]):0;
  });
  const next=Math.max(saved,...used,0)+1;
  localStorage.setItem(`brcondos_receipt_last_${year}`,String(next));
  return `${String(next).padStart(4,'0')}/${year}`;
}
function syncBillingFromBoletos(){
  const source=boletos.filter(b=>b.competence && b.section!=='CONTABIL' && !/CONTABIL/i.test(String(b.description||'')));
  let addedNfse=0,addedReceipts=0,ignored=0,missingClient=0,converted=0,locked=0;

  source.forEach((b,i)=>{
    const c=clients.find(x=>x.id===b.clientId)||findClientByLooseName(b.client);
    if(!c){missingClient++;return;}

    const nfExisting=nfse.find(x=>String(x.sourceBoletoId)===String(b.id));
    const recExisting=receipts.find(x=>String(x.sourceBoletoId)===String(b.id));

    if(c.billingDocument==='receipt'){
      if(recExisting){ignored++;return;}
      if(nfExisting){
        if(['emitida_nfse','cancelada_nfse'].includes(nfExisting.status)){locked++;return;}
        nfse=nfse.filter(x=>x.id!==nfExisting.id);
        converted++;
      }

      const issueDate=receiptDefaultIssueDate(b.competence,b.due||'');
      const year=String(b.competence||issueDate).slice(0,4);
      receipts.push(normalizeReceipt({
        id:Date.now()+i+Math.floor(Math.random()*100),
        sourceBoletoId:b.id,
        clientId:c.id,
        client:c.name||b.client,
        competence:b.competence,
        issueDate,
        value:Number(b.value||0),
        description:'Honorários administrativos',
        receiptNumber:nextReceiptNumber(year),
        status:'pendente'
      }));
      addedReceipts++;
      return;
    }

    if(nfExisting){ignored++;return;}
    if(recExisting){
      if(recExisting.status==='gerado'){locked++;return;}
      receipts=receipts.filter(x=>x.id!==recExisting.id);
      converted++;
    }

    nfse.push({
      id:Date.now()+i+Math.floor(Math.random()*100),sourceBoletoId:b.id,clientId:c.id||b.clientId||0,client:b.client,
      competence:b.competence,value:Number(b.value||0),description:'Administração de Condomínios.',
      rpsNumber:nextRpsNumber(),issueDate:today(),serviceDate:'',aliquotaPct:null,status:'rascunho',
      nfseNumber:'',verificationCode:'',gissProtocol:'',lastError:'',gissResponse:null
    });
    addedNfse++;
  });

  saveData('nfse',nfse);
  saveData('receipts',receipts);
  renderAll();
  alert(
    `Sincronização concluída.\n\n`+
    `NFS-e criadas: ${addedNfse}\n`+
    `Recibos criados: ${addedReceipts}\n`+
    `Convertidos após mudança no cadastro: ${converted}\n`+
    `Já sincronizados: ${ignored}\n`+
    `Cliente não encontrado: ${missingClient}\n`+
    `Documentos já emitidos/gerados e preservados: ${locked}\n\n`+
    `O sistema separou automaticamente conforme o campo Documento de faturamento do cadastro de Clientes.`
  );
}

  syncBillingFromBoletos();
}

  }
  const description=val('nf_desc');

  nfse=nfse.map(x=>x.id===id?{
